1//! Brushable block behavior for suspicious sand and suspicious gravel.

5use steel_macros::block_behavior;

9use steel_registry::sound_event::SoundEventRef;

19/// Vanilla archaeology block behavior for suspicious sand and suspicious gravel.
20#[block_behavior]
21pub struct BrushableBlock {
22    block: BlockRef,
23    #[json_arg(vanilla_blocks, json = "turns_into")]
24    turns_into: BlockRef,
25    #[json_arg(sound_events, json = "brush_sound")]
26    brush_sound: SoundEventRef,
27    #[json_arg(sound_events, json = "brush_completed_sound")]
28    brush_completed_sound: SoundEventRef,
29}
30
31const DUSTED: &IntProperty = &BlockStateProperties::DUSTED;
32
33impl BrushableBlock {
34    /// Creates a brushable block behavior from extracted vanilla block arguments.
35    #[must_use]
36    pub const fn new(
37        block: BlockRef,
38        turns_into: BlockRef,
39        brush_sound: SoundEventRef,
40        brush_completed_sound: SoundEventRef,
41    ) -> Self {
42        Self {
43            block,
44            turns_into,
45            brush_sound,
46            brush_completed_sound,
47        }
48    }
49}
50
51impl BlockBehavior for BrushableBlock {
52    fn get_state_for_placement(&self, _context: &BlockPlaceContext<'_>) -> Option<BlockStateId> {
53        Some(self.block.default_state().set_value(DUSTED, 0))
54    }
55
56    fn on_place(
57        &self,
58        _state: BlockStateId,
59        world: &Arc<World>,
60        pos: BlockPos,
61        _old_state: BlockStateId,
62        _moved_by_piston: bool,
63    ) {
64        world.schedule_block_tick_default(pos, self.block, 2);
65    }
66
67    fn update_shape(
68        &self,
69        state: BlockStateId,
70        world: &dyn ScheduledTickAccess,
71        pos: BlockPos,
72        _direction: Direction,
73        _neighbor_pos: BlockPos,
74        _neighbor_state: BlockStateId,
75    ) -> BlockStateId {
76        let _ = world.schedule_block_tick_default(pos, self.block, 2);
77        state
78    }
79
80    fn tick(&self, _state: BlockStateId, world: &Arc<World>, pos: BlockPos) {
81        let Some(block_entity) = world.get_block_entity(pos) else {
82            return;
83        };
84        let Some(brushable) = block_entity.downcast_ref::<BrushableBlockEntity>() else {
85            return;
86        };
87        let mutation = brushable.check_reset(world);
88        mutation.apply(world, pos);
89
90        // TODO: Fall without dropping once Steel implements vanilla falling-block entities.
91    }
92
93    fn new_block_entity(
94        &self,
95        level: Weak<World>,
96        pos: BlockPos,
97        state: BlockStateId,
98    ) -> BlockEntityCreation {
99        BlockEntityCreation::from_registered_factory(BLOCK_ENTITIES.create(
100            &vanilla_block_entity_types::BRUSHABLE_BLOCK,
101            level,
102            pos,
103            state,
104        ))
105    }
106
107    fn should_keep_block_entity(&self, old_state: BlockStateId, new_state: BlockStateId) -> bool {
108        old_state.get_block() == new_state.get_block()
109    }
110
111    fn brushable_data(&self, _state: BlockStateId) -> Option<BrushableData> {
112        Some(BrushableData {
113            turns_into: self.turns_into,
114            brush_sound: self.brush_sound,
115            brush_completed_sound: self.brush_completed_sound,
116        })
117    }
118}
