21/// Vanilla `ChorusPlantBlock` connection and survival behavior.
22// TODO: Implement ticking and full shape-update side effects.

24pub struct ChorusPlantBlock {

26}

28const DOWN: &BoolProperty = &BlockStateProperties::DOWN;
29const EAST: &BoolProperty = &BlockStateProperties::EAST;
30const NORTH: &BoolProperty = &BlockStateProperties::NORTH;
31const SOUTH: &BoolProperty = &BlockStateProperties::SOUTH;
32const UP: &BoolProperty = &BlockStateProperties::UP;
33const WEST: &BoolProperty = &BlockStateProperties::WEST;

35impl ChorusPlantBlock {
36    /// Creates a new chorus plant block behavior.
37    #[must_use]
38    pub const fn new(block: BlockRef) -> Self {
39        Self { block }
40    }
41
42    #[must_use]
43    pub(crate) fn state_with_connections(
44        world: &dyn LevelReader,
45        pos: BlockPos,
46        mut state: BlockStateId,
47    ) -> BlockStateId {
48        let down = world.get_block_state(pos.below());
49        let up = world.get_block_state(pos.above());
50        let north = world.get_block_state(pos.north());
51        let east = world.get_block_state(pos.east());
52        let south = world.get_block_state(pos.south());
53        let west = world.get_block_state(pos.west());
54        let block = state.get_block();
55
56        state = state.set_value(
57            DOWN,
58            down.get_block() == block
59                || down.get_block() == &vanilla_blocks::CHORUS_FLOWER
60                || down.get_block().has_tag(&BlockTag::SUPPORTS_CHORUS_PLANT),
61        );
62        state = state.set_value(
63            UP,
64            up.get_block() == block || up.get_block() == &vanilla_blocks::CHORUS_FLOWER,
65        );
66        state = state.set_value(
67            NORTH,
68            north.get_block() == block || north.get_block() == &vanilla_blocks::CHORUS_FLOWER,
69        );
70        state = state.set_value(
71            EAST,
72            east.get_block() == block || east.get_block() == &vanilla_blocks::CHORUS_FLOWER,
73        );
74        state = state.set_value(
75            SOUTH,
76            south.get_block() == block || south.get_block() == &vanilla_blocks::CHORUS_FLOWER,
77        );
78        state.set_value(
79            WEST,
80            west.get_block() == block || west.get_block() == &vanilla_blocks::CHORUS_FLOWER,
81        )
82    }
83}

85impl BlockBehavior for ChorusPlantBlock {
86    fn can_survive(&self, _state: BlockStateId, world: &dyn LevelReader, pos: BlockPos) -> bool {
87        let below_state = world.get_block_state(pos.below());
88        let block_above_or_below =
89            !world.get_block_state(pos.above()).is_air() && !below_state.is_air();
90
91        for direction in HORIZONTAL_DIRECTIONS {
92            let neighbor_pos = pos.relative(direction);
93            let neighbor_state = world.get_block_state(neighbor_pos);
94            if neighbor_state.get_block() == self.block {
95                if block_above_or_below {
96                    return false;
97                }
98
99                let below = world.get_block_state(neighbor_pos.below());
100                if below.get_block() == self.block
101                    || below.get_block().has_tag(&BlockTag::SUPPORTS_CHORUS_PLANT)
102                {
103                    return true;
104                }
105            }
106        }
107
108        below_state.get_block() == self.block
109            || below_state
110                .get_block()
111                .has_tag(&BlockTag::SUPPORTS_CHORUS_PLANT)
112    }
113
114    fn get_state_for_placement(&self, context: &BlockPlaceContext<'_>) -> Option<BlockStateId> {
115        Some(Self::state_with_connections(
116            context.world.as_ref(),
117            context.place_pos(),
118            self.block.default_state(),
119        ))
120    }
