25/// Vanilla `RedStoneWireBlock` using `DefaultRedstoneWireEvaluator`.
26///
27/// Steel intentionally does not implement the experimental redstone feature.
28/// Signal suppression is carried by [`SignalQueryContext`] instead of mutating
29/// vanilla's process-global `shouldSignal` field.
30#[block_behavior]
31pub struct RedStoneWireBlock {
32    block: BlockRef,
33    cross_state: BlockStateId,
34    evaluator: DefaultRedstoneWireEvaluator,
35}
36
37const EAST_REDSTONE: &EnumProperty<RedstoneSide> = &BlockStateProperties::EAST_REDSTONE;
38const FACING: &EnumProperty<Direction> = &BlockStateProperties::FACING;
39const HORIZONTAL_FACING: &EnumProperty<Direction> = &BlockStateProperties::HORIZONTAL_FACING;
40const NORTH_REDSTONE: &EnumProperty<RedstoneSide> = &BlockStateProperties::NORTH_REDSTONE;
41const POWER: &IntProperty = &BlockStateProperties::POWER;
42const SOUTH_REDSTONE: &EnumProperty<RedstoneSide> = &BlockStateProperties::SOUTH_REDSTONE;
43const WEST_REDSTONE: &EnumProperty<RedstoneSide> = &BlockStateProperties::WEST_REDSTONE;
44
45impl RedStoneWireBlock {
46    /// Creates the ordinary redstone-wire behavior and its persistent evaluator.
47    #[must_use]
48    pub fn new(block: BlockRef) -> Self {
49        let cross_state = block
50            .default_state()
51            .set_value(NORTH_REDSTONE, RedstoneSide::Side)
52            .set_value(EAST_REDSTONE, RedstoneSide::Side)
53            .set_value(SOUTH_REDSTONE, RedstoneSide::Side)
54            .set_value(WEST_REDSTONE, RedstoneSide::Side);
55        Self {
56            block,
57            cross_state,
58            evaluator: DefaultRedstoneWireEvaluator::new(block),
59        }
60    }
61
62    const fn property_for_direction(
63        direction: Direction,
64    ) -> Option<&'static EnumProperty<RedstoneSide>> {
65        match direction {
66            Direction::North => Some(NORTH_REDSTONE),
67            Direction::East => Some(EAST_REDSTONE),
68            Direction::South => Some(SOUTH_REDSTONE),
69            Direction::West => Some(WEST_REDSTONE),
70            Direction::Down | Direction::Up => None,
71        }
72    }
73
74    fn is_connected(side: RedstoneSide) -> bool {
75        side != RedstoneSide::None
76    }
77
78    fn is_cross(state: BlockStateId) -> bool {
79        Direction::HORIZONTAL.into_iter().all(|direction| {
80            Self::property_for_direction(direction)
81                .is_some_and(|property| Self::is_connected(state.get_value(property)))
82        })
83    }
84
85    fn is_dot(state: BlockStateId) -> bool {
86        Direction::HORIZONTAL.into_iter().all(|direction| {
87            Self::property_for_direction(direction)
88                .is_some_and(|property| !Self::is_connected(state.get_value(property)))
89        })
90    }
91
92    fn get_connection_state(
93        &self,
94        level: &dyn LevelReader,
95        state: BlockStateId,
96        pos: BlockPos,
97    ) -> BlockStateId {
98        let was_dot = Self::is_dot(state);
99        let mut state = self.get_missing_connections(
100            level,
101            self.block
102                .default_state()
103                .set_value(POWER, state.get_value(POWER)),
104            pos,
105        );
106        if was_dot && Self::is_dot(state) {
107            return state;
108        }
109
110        let north = Self::is_connected(state.get_value(NORTH_REDSTONE));
111        let south = Self::is_connected(state.get_value(SOUTH_REDSTONE));
112        let east = Self::is_connected(state.get_value(EAST_REDSTONE));
113        let west = Self::is_connected(state.get_value(WEST_REDSTONE));
114        let north_south_empty = !north && !south;
115        let east_west_empty = !east && !west;
116
117        if !west && north_south_empty {
118            state = state.set_value(WEST_REDSTONE, RedstoneSide::Side);
119        }
120        if !east && north_south_empty {
121            state = state.set_value(EAST_REDSTONE, RedstoneSide::Side);
122        }
123        if !north && east_west_empty {
124            state = state.set_value(NORTH_REDSTONE, RedstoneSide::Side);
125        }
126        if !south && east_west_empty {
127            state = state.set_value(SOUTH_REDSTONE, RedstoneSide::Side);
128        }
129
130        state
131    }
132
133    fn get_missing_connections(
134        &self,
135        level: &dyn LevelReader,
136        mut state: BlockStateId,
137        pos: BlockPos,
138    ) -> BlockStateId {
139        let above_state = level.get_block_state(pos.above());
140        // Vanilla passes the wire position, rather than `pos.above()`, to this
141        // state predicate in `getMissingConnections`.
142        let can_connect_up = !is_redstone_conductor(level, above_state, pos);
143
144        for direction in Direction::HORIZONTAL {
145            let Some(property) = Self::property_for_direction(direction) else {
146                continue;
147            };
148            if !Self::is_connected(state.get_value(property)) {
149                state = state.set_value(
150                    property,
151                    self.get_connecting_side_with_up(level, pos, direction, can_connect_up),
152                );
153            }
154        }
155
156        state
157    }
158
159    fn get_connecting_side(
160        &self,
161        level: &dyn LevelReader,
162        pos: BlockPos,
163        direction: Direction,
164    ) -> RedstoneSide {
165        let above_pos = pos.above();
166        let can_connect_up = !is_redstone_conductor(level, level.get_block_state(above_pos), pos);
167        self.get_connecting_side_with_up(level, pos, direction, can_connect_up)
168    }
169
170    fn get_connecting_side_with_up(
171        &self,
172        level: &dyn LevelReader,
173        pos: BlockPos,
174        direction: Direction,
175        can_connect_up: bool,
176    ) -> RedstoneSide {
177        let relative_pos = pos.relative(direction);
178        let relative_state = level.get_block_state(relative_pos);
179
180        if can_connect_up {
181            let behavior = BLOCK_BEHAVIORS.get_behavior(relative_state.get_block());
182            let is_placeable_above =
183                behavior.is_trapdoor() || Self::can_survive_on(level, relative_pos, relative_state);
184            if is_placeable_above
185                && self.should_connect_to(level.get_block_state(relative_pos.above()), None)
186            {
187                if level.is_face_sturdy(relative_state, relative_pos, direction.opposite()) {
188                    return RedstoneSide::Up;
189                }
190                return RedstoneSide::Side;
191            }
192        }
193
194        if !self.should_connect_to(relative_state, Some(direction))
195            && (is_redstone_conductor(level, relative_state, relative_pos)
196                || !self.should_connect_to(level.get_block_state(relative_pos.below()), None))
197        {
198            RedstoneSide::None
199        } else {
200            RedstoneSide::Side
201        }
202    }
203
204    fn should_connect_to(&self, state: BlockStateId, direction: Option<Direction>) -> bool {
205        if state.get_block() == self.block {
206            return true;
207        }
208        if state.get_block() == &vanilla_blocks::REPEATER {
209            let facing = state.get_value(HORIZONTAL_FACING);
210            return direction == Some(facing) || direction == Some(facing.opposite());
211        }
212        if state.get_block() == &vanilla_blocks::OBSERVER {
213            return direction == Some(state.get_value(FACING));
214        }
215
216        direction.is_some()
217            && BLOCK_BEHAVIORS
218                .get_behavior(state.get_block())
219                .is_signal_source(state, SignalQueryContext::DEFAULT)
220    }
221
222    fn can_survive_on(level: &dyn LevelReader, pos: BlockPos, state: BlockStateId) -> bool {
223        level.is_face_sturdy(state, pos, Direction::Up)
224            || state.get_block() == &vanilla_blocks::HOPPER
225    }
226
227    fn check_corner_change_at(&self, world: &Arc<World>, pos: BlockPos) {
228        if world.get_block_state(pos).get_block() != self.block {
229            return;
230        }
231
232        world.update_neighbors_at(pos, self.block);
233        for direction in Direction::ALL {
234            world.update_neighbors_at(pos.relative(direction), self.block);
235        }
236    }
237
238    fn update_neighbors_of_neighboring_wires(&self, world: &Arc<World>, pos: BlockPos) {
239        for direction in Direction::HORIZONTAL {
240            self.check_corner_change_at(world, pos.relative(direction));
241        }
242
243        for direction in Direction::HORIZONTAL {
244            let target = pos.relative(direction);
245            let target_state = world.get_block_state(target);
246            if is_redstone_conductor(world.as_ref(), target_state, target) {
247                self.check_corner_change_at(world, target.above());
248            } else {
249                self.check_corner_change_at(world, target.below());
250            }
251        }
252    }
253
254    fn updates_on_shape_change(
255        world: &Arc<World>,
256        pos: BlockPos,
257        old_state: BlockStateId,
258        new_state: BlockStateId,
259    ) {
260        for direction in Direction::HORIZONTAL {
261            let Some(property) = Self::property_for_direction(direction) else {
262                continue;
263            };
264            if Self::is_connected(old_state.get_value(property))
265                == Self::is_connected(new_state.get_value(property))
266            {
267                continue;
268            }
269
270            let relative_pos = pos.relative(direction);
271            let relative_state = world.get_block_state(relative_pos);
272            if is_redstone_conductor(world.as_ref(), relative_state, relative_pos) {
273                world.update_neighbors_at_except_from_facing(
274                    relative_pos,
275                    new_state.get_block(),
276                    direction.opposite(),
277                );
278            }
279        }
280    }
281}

283impl BlockBehavior for RedStoneWireBlock {
284    fn get_state_for_placement(&self, context: &BlockPlaceContext<'_>) -> Option<BlockStateId> {
285        Some(self.get_connection_state(
286            context.world.as_ref(),
287            self.cross_state,
288            context.place_pos(),
289        ))
290    }
291
292    fn update_shape(
293        &self,
294        state: BlockStateId,
295        world: &dyn ScheduledTickAccess,
296        pos: BlockPos,
297        direction: Direction,
298        neighbor_pos: BlockPos,
299        neighbor_state: BlockStateId,
300    ) -> BlockStateId {
301        if direction == Direction::Down {
302            return if Self::can_survive_on(world, neighbor_pos, neighbor_state) {
303                state
304            } else {
305                REGISTRY.blocks.get_default_state_id(&vanilla_blocks::AIR)
306            };
307        }
308        if direction == Direction::Up {
309            return self.get_connection_state(world, state, pos);
310        }
311
312        let Some(property) = Self::property_for_direction(direction) else {
313            return state;
314        };
315        let side_connection = self.get_connecting_side(world, pos, direction);
316        if Self::is_connected(side_connection) == Self::is_connected(state.get_value(property))
317            && !Self::is_cross(state)
318        {
319            state.set_value(property, side_connection)
320        } else {
321            self.get_connection_state(
322                world,
323                self.cross_state
324                    .set_value(POWER, state.get_value(POWER))
325                    .set_value(property, side_connection),
326                pos,
327            )
328        }
329    }
330
331    fn update_indirect_neighbour_shapes(
332        &self,
333        state: BlockStateId,
334        world: &Arc<World>,
335        pos: BlockPos,
336        flags: UpdateFlags,
337        update_limit: i32,
338    ) {
339        for direction in Direction::HORIZONTAL {
340            let Some(property) = Self::property_for_direction(direction) else {
341                continue;
342            };
343            if !Self::is_connected(state.get_value(property)) {
344                continue;
345            }
346
347            let adjacent_pos = pos.relative(direction);
348            if world.get_block_state(adjacent_pos).get_block() == self.block {
349                continue;
350            }
351
352            let below_pos = adjacent_pos.below();
353            if world.get_block_state(below_pos).get_block() == self.block {
354                let neighbor_pos = below_pos.relative(direction.opposite());
355                world.neighbor_shape_changed(
356                    direction.opposite(),
357                    below_pos,
358                    neighbor_pos,
359                    world.get_block_state(neighbor_pos),
360                    flags,
361                    update_limit,
362                );
363            }
364
365            let above_pos = adjacent_pos.above();
366            if world.get_block_state(above_pos).get_block() == self.block {
367                let neighbor_pos = above_pos.relative(direction.opposite());
368                world.neighbor_shape_changed(
369                    direction.opposite(),
370                    above_pos,
371                    neighbor_pos,
372                    world.get_block_state(neighbor_pos),
373                    flags,
374                    update_limit,
375                );
376            }
377        }
378    }
379
380    fn can_survive(&self, _state: BlockStateId, world: &dyn LevelReader, pos: BlockPos) -> bool {
381        let below_pos = pos.below();
382        Self::can_survive_on(world, below_pos, world.get_block_state(below_pos))
383    }
384
385    fn on_place(
386        &self,
387        state: BlockStateId,
388        world: &Arc<World>,
389        pos: BlockPos,
390        old_state: BlockStateId,
391        _moved_by_piston: bool,
392    ) {
393        if old_state.get_block() == self.block {
394            return;
395        }
396
397        self.evaluator.update_power_strength(world, pos, state);
398        for direction in [Direction::Down, Direction::Up] {
399            world.update_neighbors_at(pos.relative(direction), self.block);
400        }
401        self.update_neighbors_of_neighboring_wires(world, pos);
402    }
403
404    fn affect_neighbors_after_removal(
405        &self,
406        state: BlockStateId,
407        world: &Arc<World>,
408        pos: BlockPos,
409        moved_by_piston: bool,
410    ) {
411        if moved_by_piston {
412            return;
413        }
414
415        for direction in Direction::ALL {
416            world.update_neighbors_at(pos.relative(direction), self.block);
417        }
418        self.evaluator.update_power_strength(world, pos, state);
419        self.update_neighbors_of_neighboring_wires(world, pos);
420    }
421
422    fn handle_neighbor_changed(
423        &self,
424        state: BlockStateId,
425        world: &Arc<World>,
426        pos: BlockPos,
427        _source_block: BlockRef,
428        _moved_by_piston: bool,
429    ) {
430        if self.can_survive(state, world.as_ref(), pos) {
431            self.evaluator.update_power_strength(world, pos, state);
432        } else {
433            world.drop_resources(state, pos);
434            world.remove_block(pos, false);
435        }
436    }
437
438    fn use_without_item(
439        &self,
440        state: BlockStateId,
441        world: &Arc<World>,
442        pos: BlockPos,
443        player: &Player,
444        _hit_result: &BlockHitResult,
445        _inv: &mut InventoryAccess,
446    ) -> InteractionResult {
447        if !player.abilities.lock().may_build {
448            return InteractionResult::Pass;
449        }
450        if !Self::is_cross(state) && !Self::is_dot(state) {
451            return InteractionResult::Pass;
452        }
453
454        let new_base_state = if Self::is_cross(state) {
455            self.block.default_state()
456        } else {
457            self.cross_state
458        };
459        let new_state = self.get_connection_state(
460            world.as_ref(),
461            new_base_state.set_value(POWER, state.get_value(POWER)),
462            pos,
463        );
464        if new_state == state {
465            return InteractionResult::Pass;
466        }
467
468        world.set_block(pos, new_state, UpdateFlags::UPDATE_ALL);
469        Self::updates_on_shape_change(world, pos, state, new_state);
470        InteractionResult::Success
471    }
472
473    fn is_signal_source(&self, _state: BlockStateId, context: SignalQueryContext) -> bool {
474        context.wire_signals_enabled()
475    }
476
477    fn get_own_signal(
478        &self,
479        state: BlockStateId,
480        _world: &dyn LevelReader,
481        _pos: BlockPos,
482        _context: SignalQueryContext,
483    ) -> i32 {
484        i32::from(state.get_value(POWER))
485    }
486
487    fn get_signal(
488        &self,
489        state: BlockStateId,
490        world: &dyn LevelReader,
491        pos: BlockPos,
492        direction: Direction,
493        context: SignalQueryContext,
494    ) -> i32 {
495        if !context.wire_signals_enabled() || direction == Direction::Down {
496            return 0;
497        }
498
499        let power = self.get_own_signal(state, world, pos, context);
500        if power == 0 {
501            return 0;
502        }
503        if direction == Direction::Up {
504            return power;
505        }
506
507        let Some(property) = Self::property_for_direction(direction.opposite()) else {
508            return 0;
509        };
510        if Self::is_connected(
511            self.get_connection_state(world, state, pos)
512                .get_value(property),
513        ) {
514            power
515        } else {
516            0
517        }
518    }
519
520    fn get_direct_signal(
521        &self,
522        state: BlockStateId,
523        world: &dyn LevelReader,
524        pos: BlockPos,
525        direction: Direction,
526        context: SignalQueryContext,
527    ) -> i32 {
528        if context.wire_signals_enabled() {
529            self.get_signal(state, world, pos, direction, context)
530        } else {
531            0
532        }
533    }

553    #[test]
554    fn isolated_cross_and_dot_preserve_their_vanilla_shapes() {
555        let behavior = wire();
556        let level = TestLevel::default().with_block(
557            BlockPos::new(0, 63, 0),
558            vanilla_blocks::STONE.default_state(),
559        );
560        let pos = BlockPos::new(0, 64, 0);
561
562        let cross = behavior.get_connection_state(&level, behavior.cross_state, pos);
563        let dot = behavior.get_connection_state(
564            &level,
565            vanilla_blocks::REDSTONE_WIRE.default_state(),
566            pos,
567        );
568
569        assert!(RedStoneWireBlock::is_cross(cross));
570        assert!(RedStoneWireBlock::is_dot(dot));
571    }

573    #[test]
574    fn wire_climbs_sturdy_neighbor_only_when_above_is_connectable() {
575        let behavior = wire();
576        let pos = BlockPos::new(0, 64, 0);
577        let level = TestLevel::default()
578            .with_block(pos.below(), vanilla_blocks::STONE.default_state())
579            .with_block(pos.east(), vanilla_blocks::STONE.default_state())
580            .with_block(pos.east().above(), behavior.block.default_state());
581
582        assert_eq!(
583            behavior.get_connecting_side(&level, pos, Direction::East),
584            RedstoneSide::Up
585        );
586
587        level.set_test_block(pos.above(), vanilla_blocks::STONE.default_state());
588        assert_eq!(
589            behavior.get_connecting_side(&level, pos, Direction::East),
590            RedstoneSide::None
591        );
592    }

594    #[test]
595    fn powered_wire_signal_follows_recomputed_connections() {
596        let behavior = wire();
597        let pos = BlockPos::new(0, 64, 0);
598        let state = behavior
599            .block
600            .default_state()
601            .set_value(POWER, 9)
602            .set_value(NORTH_REDSTONE, RedstoneSide::Side)

604        let level = TestLevel::default()
605            .with_block(pos.below(), vanilla_blocks::STONE.default_state())
606            .with_block(pos.north(), behavior.block.default_state())
607            .with_block(pos.east(), behavior.block.default_state());

622                &level,
623                pos,
624                Direction::East,
625                SignalQueryContext::DEFAULT,

628        );

630            behavior.get_signal(
631                state,
632                &level,
633                pos,
634                Direction::Up,

636            ),
637            0
638        );
